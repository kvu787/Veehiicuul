using Godot;
using System;
using System.Diagnostics;
using System.Linq;
using System.Reflection;

namespace Veehiicuul_Godot_CSharp;

// Installed in an isolated project copy by RunNative.ps1. Historical sources are compiled
// into this measurement build only; no baseline code enters normal exports.
public partial class Main_GodotAdapter : Node {
    public override void _Ready() {
        try {
            Main main = new(this);
            main.Ready();
            CarSwitcher cars = GetField<CarSwitcher>(main, "CarSwitcher");
            CollisionManager manager = GetField<CollisionManager>(main, "CollisionManager");
            CarStateManager state = GetField<CarStateManager>(main, "CarStateManager");
            ColliderJson data = JsonUtility.Deserialize<ColliderJson>("res://Tracks/Ribeye/Ribeye_ColliderData.json");
            BaselineCollisionManager baseline = new("Ribeye", cars);
            RectangleLocalBounds[] bounds = GetField<RectangleLocalBounds[]>(manager, "_vehicleBounds");
            TrackCollisionDetector detector = new(data, bounds[cars.CurrentCarIndex]);
            Vector3 spawnPosition = state.Position;
            float spawnYaw = state.Rotation;
            PropertyInfo carIndex = typeof(CarSwitcher).GetProperty("CurrentCarIndex")!;
            int initialCar = cars.CurrentCarIndex;
            for (int index = 0; index < 6; ++index) {
                carIndex.SetValue(cars, index);
                state.Reset_PositionRotationVelocity();
                state.Apply();
                Require(!manager.IsCarColliding(state.Position, state.Rotation), "Spawn collides");
                Require(!baseline.IsCarColliding(), "Baseline spawn collides");
                GD.Print($"SPAWN {cars.CurrentCar.GameObjectName}: clear; bounds={bounds[index].MinX:R},{bounds[index].MinY:R},{bounds[index].MaxX:R},{bounds[index].MaxY:R}");
            }
            carIndex.SetValue(cars, initialCar);
            state.Apply();
            Node3D vehicle = cars.CurrentCar.Node!;
            CoordinateXY vertex = data.Outlines[0].Vertices[0];
            Vector3 contactPosition = new(vertex.X, 0, -vertex.Y);
            const float contactYaw = 0.3f;
            Require(manager.IsCarColliding(contactPosition, contactYaw), "Expected contact");
            Measure("OptimizedStationarySpawn", () => manager.IsCarColliding(spawnPosition, spawnYaw));
            Measure("BaselineStationarySpawn", baseline.IsCarColliding);
            Measure("OptimizedStationaryContact", () => manager.IsCarColliding(contactPosition, contactYaw));
            vehicle.Position = contactPosition;
            vehicle.Rotation = new(0, contactYaw, 0);
            Measure("BaselineStationaryContact", baseline.IsCarColliding);
            Random random = new(20260929);
            Vector3[] positions = new Vector3[4096];
            float[] angles = new float[4096];
            for (int index = 0; index < positions.Length; ++index) {
                positions[index] = new Vector3((float)random.NextDouble() * 400 - 200, 0, (float)random.NextDouble() * 400 - 200);
                angles[index] = (float)random.NextDouble() * 6 - 3;
            }
            int comparisons = 0, legacyMismatches = 0;
            for (int selected = 0; selected < 6; ++selected) {
                carIndex.SetValue(cars, selected);
                vehicle = cars.CurrentCar.Node!;
                for (int index = 0; index < positions.Length; ++index) {
                    bool expected = detector.IsCollidingLinear(bounds[selected], new(positions[index].X, -positions[index].Z, -angles[index]));
                    bool actual = manager.IsCarColliding(positions[index], angles[index]);
                    Require(actual == expected, "Native moving query mismatch");
                    Require(manager.IsCarColliding(positions[index], angles[index]) == expected, "Cached query mismatch");
                    vehicle.Position = positions[index];
                    vehicle.Rotation = new(0, angles[index], 0);
                    if (baseline.IsCarColliding() != actual) { ++legacyMismatches; }
                    ++comparisons;
                }
            }
            Require(legacyMismatches == 0, "Legacy/new native random-pose mismatch");
            GD.Print($"PASS: {comparisons} moving poses across six cars, {comparisons} cached repeats, zero historical native mismatches.");
            carIndex.SetValue(cars, initialCar);
            vehicle = cars.CurrentCar.Node!;
            int cursor = 0;
            Measure("OptimizedMovingQuery", () => {
                int index = cursor++ & 4095;
                return manager.IsCarColliding(positions[index], angles[index]);
            });
            cursor = 0;
            Measure("OptimizedApplyAndQuery", () => {
                int index = cursor++ & 4095;
                vehicle.Position = positions[index];
                vehicle.Rotation = new(0, angles[index], 0);
                return manager.IsCarColliding(positions[index], angles[index]);
            });
            cursor = 0;
            Measure("BaselineApplyAndQuery", () => {
                int index = cursor++ & 4095;
                vehicle.Position = positions[index];
                vehicle.Rotation = new(0, angles[index], 0);
                return baseline.IsCarColliding();
            });
            cursor = 0;
            Measure("ApplyOnly", () => {
                int index = cursor++ & 4095;
                vehicle.Position = positions[index];
                vehicle.Rotation = new(0, angles[index], 0);
                return false;
            });
            Vector3[] contacts = data.Outlines.ConvertAll(outline => outline.Vertices).SelectMany(points => points)
                .Select(point => new Vector3(point.X, 0, -point.Y)).ToArray();
            foreach (Vector3 position in contacts) {
                vehicle.Position = position;
                vehicle.Rotation = new(0, contactYaw, 0);
                Require(manager.IsCarColliding(position, contactYaw) == baseline.IsCarColliding(), "Native boundary-pose mismatch");
            }
            GD.Print($"PASS: {contacts.Length} native track-boundary poses match the historical implementation.");
            cursor = 0;
            Measure("OptimizedContactApplyAndQuery", () => {
                Vector3 position = contacts[cursor++ % contacts.Length];
                vehicle.Position = position;
                vehicle.Rotation = new(0, contactYaw, 0);
                return manager.IsCarColliding(position, contactYaw);
            });
            cursor = 0;
            Measure("BaselineContactApplyAndQuery", () => {
                Vector3 position = contacts[cursor++ % contacts.Length];
                vehicle.Position = position;
                vehicle.Rotation = new(0, contactYaw, 0);
                return baseline.IsCarColliding();
            });
            ValidatePoseChanges(manager, detector, bounds[cars.CurrentCarIndex], vertex);
            ValidateDistinctCarBounds(cars, carIndex, data, contactPosition);
            ValidateFootprintHierarchy();
            carIndex.SetValue(cars, initialCar);
            state.Reset_PositionRotationVelocity();
            state.Apply();
            typeof(CarStateManager).GetProperty("Position")!.SetValue(state, contactPosition);
            typeof(CarStateManager).GetField("Rotation_ForMostRecentNonZeroVelocity", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(state, contactYaw);
            state.Apply();
            Require(manager.IsCarColliding(state.Position, state.Rotation), "Expected contact before Main.Process");
            main.Process(0);
            DigitalInputCounts.ClearFrameCounts();
            Require(state.Position == spawnPosition && state.Rotation == spawnYaw, "Main.Process failed to reset collision");
            DigitalInputCounts digital = this.GetNode<DigitalInputCounts>("/root/DigitalInputCounts");
            Press(main, digital, JoyButton.DpadRight);
            Require(cars.CurrentCarIndex != initialCar && state.Position == spawnPosition, "Car switch did not reset");
            typeof(CarStateManager).GetProperty("Position")!.SetValue(state, new Vector3(10000, 0, 10000));
            state.Apply();
            Press(main, digital, JoyButton.X);
            Require(state.Position == spawnPosition && !manager.IsCarColliding(state.Position, state.Rotation), "Explicit reset failed");
            Press(main, digital, JoyButton.DpadUp);
            CollisionManager reloadedManager = GetField<CollisionManager>(main, "CollisionManager");
            CarStateManager reloadedState = GetField<CarStateManager>(main, "CarStateManager");
            Require(!ReferenceEquals(manager, reloadedManager), "Track reload reused old cache");
            Require(!reloadedManager.IsCarColliding(reloadedState.Position, reloadedState.Rotation), "Track reload spawn collided");
            GD.Print("PASS: collision reset, input-driven car switch, explicit reset, and track reload.");
            GD.Print("PASS: native collision analysis completed.");
            this.GetTree().Quit(0);
        } catch (Exception exception) {
            GD.PrintErr(exception.ToString());
            this.GetTree().Quit(1);
        }
    }

    private static void ValidatePoseChanges(CollisionManager manager, TrackCollisionDetector detector, RectangleLocalBounds bounds, CoordinateXY vertex) {
        int changes = 0;
        for (int offset = -8; offset <= 8; ++offset) {
            Vector3 position = new(vertex.X + offset * 0.5f, 0, -vertex.Y);
            for (int angle = 0; angle < 64; ++angle) {
                float yaw = angle * MathF.Tau / 64;
                bool expected = detector.IsCollidingLinear(bounds, new(position.X, -position.Z, -yaw));
                Require(manager.IsCarColliding(position, yaw) == expected, "Rotation-only cache invalidation failed");
                ++changes;
            }
        }
        GD.Print($"PASS: {changes} rotation-only and translation cache changes.");
    }

    private static void ValidateDistinctCarBounds(CarSwitcher cars, PropertyInfo carIndex, ColliderJson data, Vector3 center) {
        int originalIndex = cars.CurrentCarIndex;
        Node3D larger = cars.AvailableCars[1].Node!;
        Vector3 originalScale = larger.Scale;
        try {
            larger.Scale = originalScale * 2;
            CollisionManager sized = new("Ribeye", cars);
            RectangleLocalBounds[] bounds = GetField<RectangleLocalBounds[]>(sized, "_vehicleBounds");
            TrackCollisionDetector linear = new(data, bounds[1]);
            bool foundDifference = false;
            for (int offset = 0; offset < 64 && !foundDifference; ++offset) {
                Vector3 position = center + new Vector3(offset * 0.25f, 0, 0);
                bool first = linear.IsCollidingLinear(bounds[0], new(position.X, -position.Z, 0));
                bool second = linear.IsCollidingLinear(bounds[1], new(position.X, -position.Z, 0));
                if (first == second) { continue; }
                carIndex.SetValue(cars, 0);
                Require(sized.IsCarColliding(position, 0) == first, "Small car mismatch");
                carIndex.SetValue(cars, 1);
                Require(sized.IsCarColliding(position, 0) == second, "Car-index cache invalidation failed");
                foundDifference = true;
            }
            Require(foundDifference, "Differently sized car fixture did not distinguish results");
            GD.Print("PASS: same pose with differently sized cars invalidates cached collision.");
        } finally {
            larger.Scale = originalScale;
            carIndex.SetValue(cars, originalIndex);
        }
    }

    private static void ValidateFootprintHierarchy() {
        using Node3D root = new();
        using BoxMesh mesh = new() { Size = new Vector3(2, 2, 4) };
        MeshInstance3D child = new() { Mesh = mesh, Position = new Vector3(3, 0, -2), Visible = false };
        Node3D parent = new() { Position = new Vector3(2, 0, 1) };
        root.AddChild(parent);
        parent.AddChild(child);
        VehicleCollisionFootprint footprint = VehicleCollisionFootprint.FromMeshGeometry(root);
        Require(footprint.MinX == 4 && footprint.MaxX == 6 && footprint.MinZ == -3 && footprint.MaxZ == 1,
            "Nested hidden mesh footprint mismatch");
        child.CustomAabb = new Aabb(new Vector3(-2, -1, -1), new Vector3(4, 2, 2));
        footprint = VehicleCollisionFootprint.FromMeshGeometry(root);
        Require(footprint.MinX == 3 && footprint.MaxX == 7 && footprint.MinZ == -2 && footprint.MaxZ == 0,
            "Custom mesh footprint mismatch");
        root.Free();
        GD.Print("PASS: nested hidden meshes and custom AABBs.");
    }

    private static void Press(Main main, DigitalInputCounts digital, JoyButton button) {
        using InputEventJoypadButton press = new() { ButtonIndex = button, Pressed = true };
        digital._Input(press);
        main.Process(0);
        DigitalInputCounts.ClearFrameCounts();
    }

    private static T GetField<T>(object instance, string name) {
        return (T)instance.GetType().GetField(name, BindingFlags.NonPublic | BindingFlags.Instance)!.GetValue(instance)!;
    }

    private static void Require(bool condition, string message) {
        if (!condition) { throw new InvalidOperationException(message); }
    }

    private static void Measure(string name, Func<bool> action, int iterations = 100000) {
        int checksum = 0;
        long warmStart = Stopwatch.GetTimestamp();
        do { for (int index = 0; index < 10000; ++index) { if (action()) { ++checksum; } } }
        while (Stopwatch.GetElapsedTime(warmStart).TotalMilliseconds < 400);
        checksum = 0;
        double[] samples = new double[15];
        long before = GC.GetAllocatedBytesForCurrentThread();
        for (int sample = 0; sample < samples.Length; ++sample) {
            long start = Stopwatch.GetTimestamp();
            for (int index = 0; index < iterations; ++index) { if (action()) { ++checksum; } }
            samples[sample] = Stopwatch.GetElapsedTime(start).TotalNanoseconds / iterations;
        }
        long bytes = GC.GetAllocatedBytesForCurrentThread() - before;
        if (name.StartsWith("Optimized", StringComparison.Ordinal)) {
            Require(bytes == 0, "Optimized native queries must not allocate after warmup");
        }
        Array.Sort(samples);
        GD.Print($"NATIVE {name}: median={samples[7]:F1} ns; batch-p95={samples[14]:F1} ns; bytes/query={bytes / (15.0 * iterations):F1}; checksum={checksum}; iterations/sample={iterations}");
    }
}
